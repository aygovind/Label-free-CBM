INITIAL SETUP

git clone https://github.com/Trustworthy-ML-Lab/Label-free-CBM.git
cd Label-free-CBM

conda create -p ~/envs/lfcbm python=3.9 -y
source /opt/conda/etc/profile.d/conda.sh
conda activate ~/envs/lfcbm
~/envs/lfcbm/bin/pip install ipykernel
~/envs/lfcbm/bin/python -m ipykernel install --user --name lfcbm --display-name "Python (lfcbm)"

pip install -r requirements.txt
bash download_cub.sh
bash download_models.sh
bash download_rn18_places.sh
pip uninstall -y numpy
pip install numpy==1.26.4

RESTART

cd ~/Label-free-CBM
source /opt/conda/etc/profile.d/conda.sh
conda activate ~/envs/lfcbm
clear

TRAIN MODEL

python train_cbm.py --backbone bioclip --feature_layer visual.ln_post \
    --dataset treeoflife --concept_set data/concept_sets/treeoflife_concepts.txt

DATASET DOWNLOAD

pip install kaggle
mkdir -p ~/.kaggle && chmod 600 ~/.kaggle/kaggle.json  # after placing your token

kaggle datasets download -d gpiosenka/100-bird-species
unzip 100-bird-species.zip -d birds_data
